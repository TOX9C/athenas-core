```typescript
export function spawn(
  id: string,
  cwd: string,
  shell: string,
  agentCmd: string | undefined,
  mainWindow: BrowserWindow
): void {
  if (sessions.has(id)) {
    kill(id)
  }

  const shellPath = shell || getDefaultShell()
  const isWin = platform() === 'win32'
  
  // Use login shell flag to ensure $PATH and profiles are loaded
  const shellArgs = isWin ? [] : ['-l']

  const ptyProcess = pty.spawn(shellPath, shellArgs, {
    name: 'xterm-256color',
    cols: 80,
    rows: 24,
    cwd,
    env: process.env as Record<string, string>
  })

  sessions.set(id, ptyProcess)

  ptyProcess.onData((data) => {
    mainWindow.webContents.send(`pty:data:${id}`, data)
  })

  ptyProcess.onExit(({ exitCode }) => {
    sessions.delete(id)
    mainWindow.webContents.send(`pty:exit:${id}`, exitCode)
  })

  if (agentCmd) {
    // Longer timeout to allow shell to source profiles before writing command
    setTimeout(() => {
      if (sessions.has(id)) {
        ptyProcess.write(agentCmd + '\r')
      }
    }, 1000)
  }
}
```

// src/utils/ansi.ts
export function stripAnsi(str: string): string {
  const pattern = [
    '[\\u001B\\u009B][[\\]()#;?]*(?:(?:(?:(?:;[-a-zA-Z\\d\\/#&.:=?%@~_]+)*|[a-zA-Z\\d]+(?:;[-a-zA-Z\\d\\/#&.:=?%@~_]*)*)?\\u0007)',
    '(?:(?:\\d{1,4}(?:;\\d{0,4})*)?[\\dA-PR-TZcf-nq-uy=><~]))'
  ].join('|')
  return str.replace(new RegExp(pattern, 'g'), '')
}
```

// src/components/Athena/useAthena.ts
import { stripAnsi } from '../../utils/ansi'

// Inside useAthena hook useEffect:
  useEffect(() => {
    if (!activeSpace) return

    const unsub = window.athena.pty.onData(ATHENA_PTY_ID, (data) => {
      if (!collectingRef.current) return

      // Strip ANSI escape codes (colors, cursor movements, spinners)
      const cleanData = stripAnsi(data)
      if (cleanData) {
        bufferRef.current += cleanData
      }

      if (timeoutRef.current) clearTimeout(timeoutRef.current)
      timeoutRef.current = setTimeout(() => {
        // Only finalize message if we have meaningful text
        const content = bufferRef.current.trim()
        
        // Exclude generic shell prompts from being treated as AI answers
        const isJustPrompt = content.endsWith('%') || content.endsWith('$') || content.includes('athenas-core')
        
        if (content && !isJustPrompt) {
          addMessage({
            id: nanoid(),
            role: 'athena',
            content,
            timestamp: Date.now(),
          })
          bufferRef.current = ''
          collectingRef.current = false
        }
      }, 2000) // Increase timeout to account for slow LLM token streams
    })

    return () => {
      unsub()
      if (timeoutRef.current) clearTimeout(timeoutRef.current)
    }
  }, [activeSpace, addMessage])
```
