    94	export function toOpenAITools() {
    95	  return ORCHESTRATOR_TOOLS.map(t => ({
    96	    type: 'function' as const,
    97	    function: {
    98	      name: t.name,
    99	      description: t.description,
   100	      parameters: {
   101	        type: 'object',
   102	        properties: t.input_schema.properties,
   103	        required: t.input_schema.required
   104	      }
   105	    }
   106	  }))
   107	}
   108	
   109	function getWindow(): BrowserWindow | null {
   110	  return BrowserWindow.getAllWindows()[0] ?? null
   111	}
   112	
   113	function shellEscape(arg: string): string {
   114	  return "'" + arg.replace(/'/g, "'\\''") + "'"
   115	}
   116	
   117	function buildAgentCommand(agentType: string, taskPrompt?: string): string {
   118	  let baseCmd = 'claude'
   119	  if (agentType === 'codex') baseCmd = 'codex'
   120	  if (agentType === 'opencode') baseCmd = 'opencode'
   121	  if (agentType === 'gemini') baseCmd = 'gemini'
   122	  if (agentType === 'shell') return ''
   123	
   124	  if (!taskPrompt) return baseCmd
   125	  return `${baseCmd} -p ${shellEscape(taskPrompt)}`
   126	}
   127	
   128	export function executeToolCall(name: string, args: ToolInput): ToolCallResult {
   129	  const win = getWindow()
   130	  if (!win) {
   131	    return { text: 'Failed: No window available.' }
   132	  }
   133	
   134	  switch (name) {
   135	    case 'launch_builtin_agent': {
   136	      const { task_prompt, agent_type = 'claude', agent_count = 1 } = args
   137	      const agentCommand = buildAgentCommand(agent_type, task_prompt)
   138	      for (let i = 0; i < agent_count; i++) {
   139	        const id = `agent-${randomUUID()}`
   140	        win.webContents.send('athena:agent-spawned', { id, agentType: agent_type, agentCmd: agentCommand })
   141	      }
   142	      return { text: `Done, launched ${agent_count} ${agent_type} agents.` }
   143	    }
   144	
   145	    case 'launch_custom_agent': {
   146	      const { command, agent_count = 1 } = args
   147	      for (let i = 0; i < agent_count; i++) {
   148	        const id = `custom-agent-${randomUUID()}`
   149	        win.webContents.send('athena:agent-spawned', { id, agentType: 'custom', agentCmd: command })
   150	      }
   151	      return { text: `Done, launched ${agent_count} custom agents.` }
   152	    }
   153	
   154	    case 'close_terminals': {
   155	      const { pane_ids } = args
   156	      if (Array.isArray(pane_ids)) {
   157	        win.webContents.send('athena:close-panes', pane_ids)
   158	      }
   159	      return { text: `Closed ${pane_ids?.length ?? 0} terminal(s).` }
   160	    }
   161	
   162	    case 'run_command_in_terminals': {
   163	      const { pane_ids, command } = args
   164	      if (Array.isArray(pane_ids) && command) {
   165	        pane_ids.forEach(id => {
   166	          ptyWrite(id, command)
   167	          setTimeout(() => ptyWrite(id, '\r'), 150)
   168	        })
   169	      }
   170	      return { text: `Sent command to ${pane_ids?.length ?? 0} terminal(s).` }
   171	    }
   172	
   173	    default:
   174	      return { text: `Unknown tool: ${name}` }
   175	  }
   176	}
